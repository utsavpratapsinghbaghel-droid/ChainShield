=== src/server.js ===
'use strict';
// One server that mounts every module. Run it with: npm start
const http = require('http');
const path = require('path');
const { createRouter, readJson, sendJson, serveStatic, HttpError } = require('./lib/http');
const { createStore } = require('./store');

const modules = [
  require('./modules/m1-accounts'),
  require('./modules/m2-orders'),
  require('./modules/m3-risk'),
  require('./modules/m4-tracking'),
  require('./modules/m5-alerts'),
  require('./modules/m6-dashboard'),
  require('./modules/m8-ai'),
  require('./modules/dev'),
];

function createApp(options) {
  const opts = options || {};
  const store = createStore({ seed: opts.seed });
  const router = createRouter();
  modules.forEach((m) => m.register(router, store));
  router.get('/health', () => ({ ok: true }), { auth: false });

  const publicDir = path.join(__dirname, '..', 'public');

  const server = http.createServer(async (req, res) => {
    try {
      const url = new URL(req.url, 'http://localhost');
      const found = router.find(req.method, url.pathname);
      if (!found) {
        if (req.method === 'GET' && serveStatic(res, publicDir, url.pathname)) return;
        throw new HttpError(404, 'Not found: ' + req.method + ' ' + url.pathname);
      }

      let business = null;
      if (found.route.auth) {
        const header = req.headers.authorization || '';
        const token = header.startsWith('Bearer ') ? header.slice(7) : '';
        const id = store.sessions.get(token);
        business = id ? store.businesses.get(id) : null;
        if (!business) throw new HttpError(401, 'Log in first. Send the token as "Authorization: Bearer <token>"');
      }

      const body = req.method === 'GET' ? {} : await readJson(req);
      const result = await found.route.handler({
        params: found.params,
        query: Object.fromEntries(url.searchParams),
        body,
        business,
        store,
      });
      sendJson(res, 200, result === undefined ? { ok: true } : result);
    } catch (err) {
      const status = err.status || 500;
      if (status === 500) console.error(err);
      sendJson(res, status, { error: status === 500 ? 'Something went wrong on the server' : err.message });
    }
  });
  server.store = store;
  return server;
}

module.exports = { createApp };

if (require.main === module) {
  const port = process.env.PORT || 3000;
  createApp({ seed: true }).listen(port, () => {
    console.log('ChainShield prototype running at http://localhost:' + port);
    console.log('Demo login: mobile 9999999999, OTP 123456');
  });
}

=== src/store.js ===
'use strict';
// In-memory data store. Everything resets when the server restarts.
// Later this can be replaced by a real database (for example SAP HANA Cloud)
// without changing the modules, as long as they keep using these methods.
const crypto = require('crypto');
const path = require('path');
const fs = require('fs');

function createStore(options) {
  const store = {
    counters: {},
    sessions: new Map(),      // token -> businessId
    otps: new Map(),          // mobile -> code
    businesses: new Map(),
    partners: new Map(),
    orders: new Map(),
    risks: new Map(),         // orderId -> RiskResult[]
    alerts: new Map(),        // alertId -> Alert
    tracking: new Map(),      // orderId -> Tracking[]
    weather: [],              // demo weather overrides: { lat, lng, radiusKm, level, label }

    nextId(prefix) {
      this.counters[prefix] = (this.counters[prefix] || 0) + 1;
      return prefix + '_' + String(this.counters[prefix]).padStart(3, '0');
    },
    newToken() {
      return 'tok_' + crypto.randomBytes(12).toString('hex');
    },
    findBusinessByMobile(mobile) {
      for (const b of this.businesses.values()) if (b.mobile === mobile) return b;
      return null;
    },
  };

  if (options && options.seed) {
    const file = path.join(__dirname, '..', 'contract', 'sample-data.json');
    const sample = JSON.parse(fs.readFileSync(file, 'utf8'));
    sample.businesses.forEach((b) => store.businesses.set(b.businessId, b));
    sample.partners.forEach((p) => store.partners.set(p.partnerId, p));
    sample.orders.forEach((o) => store.orders.set(o.orderId, o));
    Object.assign(store.counters, sample.counters);
  }
  return store;
}

module.exports = { createStore };

# Ask AI with Ollama (for Virat)

The Ask AI tab talks to Ollama, a free AI that runs on your laptop. Nothing goes to the internet.
If Ollama is not running, the tab still answers with simple built-in answers, and a badge tells you which mode you are in.

## Steps (Windows, in the VS Code terminal)
1. Open the ChainShield folder in VS Code. Open a terminal: Terminal > New Terminal.
2. Check Ollama is running. Type:

       ollama list

   - If you get an error, open the **Ollama** app from the Start menu (it runs in the tray), then try again.
   - If the list has no chat model, download a small one (about 2 GB):

         ollama pull llama3.2:3b

3. Start the website:

       npm start

4. Open http://localhost:3000, log in, click **Ask AI**. The badge at the top should say **AI connected** and the model name.
   The first answer can take up to a minute because the model is loading. After that it is faster.

## Choose a model yourself (optional)
By default it picks a model that is installed. To force one, type this in PowerShell, then start:

       $env:OLLAMA_MODEL="qwen2.5:3b"
       npm start

## If the badge says "Basic answers (AI not running)"
- Ollama is not open. Start the Ollama app and refresh the page.
- Or Ollama uses a different address. Set it with: `$env:OLLAMA_URL="http://127.0.0.1:11434"`

## How the answers stay correct
Every question is sent with two things: the ChainShield fact sheet (`src/modules/m8-ai/factsheet.md`) and your own orders, suppliers and alerts.
The AI is told to answer only from these and to say "I do not know" otherwise.
If an answer is wrong, fix the wording in `factsheet.md` and restart. Do not retrain anything.

## Tell the jury
"The assistant runs locally on an open model through Ollama, so business data never leaves the machine. It is designed to move to SAP AI Core or SAP Generative AI Hub later."

# ChainShield fact sheet (the assistant may only use this and the owner's own data)

## What ChainShield is
ChainShield is an AI early-warning and backup system for Indian MSME (small manufacturer) supply chains. It was built by Team Veridians for SAP Hackfest 2026 (theme: Resilient Supply Chains). It warns the owner before a disruption such as heavy rain, a flood or a transporter strike hits, shows the money at risk in rupees, and suggests a backup supplier or a reroute.

## Two legs of every order
Leg 1 is supplier to MSME (the material arriving). Leg 2 is MSME to client (the finished order leaving). The same watching and alerts are meant to cover both legs.

## Who uses it
The MSME owner uses the full website today. Supplier and client sides are planned as the next phase. A transporter is a saved contact the owner can ask for location.

## How it works
1. The owner registers with a mobile number and OTP, then creates a password. Later logins use mobile number and password, then an OTP. The site works in Hindi or English, chosen at registration and changeable in Settings.
2. In Chain Details the owner adds suppliers, clients and transporters with name, address, phone number and an OTP.
3. The owner creates an order, choosing the client, the due date and the main suppliers with the rupee value of each part.
4. ChainShield checks the weather at each supplier's address, along the route and at the client's address.
5. If a delay would be longer than the order's buffer days, it creates an alert with the rupee value at risk and three options: switch supplier, reroute or wait. Each option shows its extra cost.
6. The owner decides. ChainShield never changes an order by itself.

## Main and alternative suppliers
Every supplier is either main or alternative. Orders use main suppliers. Alternative suppliers are kept as backups. When there is a risk, the alert suggests the best alternative supplier (calmest weather first, then closest to the client). If the owner chooses switch supplier, the order moves to that backup.

## Consent and privacy
Every supplier, client and transporter is added with their own OTP. Their OTP is their consent to receive ChainShield messages on WhatsApp. The owner also gives consent at registration for the use of mobile number and location. This follows India's DPDP Act. Passwords are stored in scrambled form. Terms and policies are linked on the login page.

## How the money numbers work
The value at risk of an order line is its value multiplied by a likelihood: 35 percent for a yellow weather alert, 65 percent for orange and 90 percent for red. It only counts when the expected delay is longer than the order's buffer (2 days unless set otherwise). Expected delays are 1, 3 and 5 days for yellow, orange and red. The extra cost of the options (6 percent of the value at risk for switching supplier, 3 percent for reroute, 0 for wait) are placeholder numbers in this prototype.

## Why an owner can trust an alert
Each alert says which place has which weather, the expected delay, the buffer, the rupee value at risk and a suggested backup. The owner reads the reason and decides.

## Driver location
On the Dashboard the owner taps Request Driver Live Location. The saved transporter is asked on WhatsApp, and if the driver approves, the location is sent. WhatsApp may only allow a one-time location pin and not a continuous live location, so a tracking link is the fallback. In this prototype the request is saved but the WhatsApp message is not sent yet.

## Plans
Freemium: free basic local weather and strike alerts. Premium: AI backup-supplier matching, rupee-value risk estimation and SAP Analytics Cloud dashboards. B2B2B partnerships: subsidised distribution through industry clusters, chambers of commerce and large enterprise buyers. Prices are not fixed yet.

## What is real today and what is planned
Real in the prototype: login with password and OTP, Hindi and English, adding suppliers, clients and transporters with OTP consent, main and alternative suppliers, orders, risk checks, alerts with a backup suggestion, switching an order to the backup, the dashboard, order history and this assistant.
Simulated or planned: the weather is set by a demo button and live weather is planned. The OTP is a fixed test code. WhatsApp and SMS sending is being connected through Meta. Data is kept in memory until it moves to SAP HANA Cloud. SAP Analytics Cloud, Joule and SAP AI Core are planned and not connected. This assistant runs on an open model through Ollama on the laptop.

## SAP technology
The app is built to run on SAP BTP (Cloud Foundry). The plan uses SAP HANA Cloud for the database, SAP CAP for the backend, and SAP Integration Suite for the weather feed and the WhatsApp connection. Connecting to an ERP such as S/4HANA is an optional extra. Joule is a future integration.

function gather(store, business) {
  const partners = [...store.partners.values()].filter((p) => p.businessId === business.businessId);
  const orders = [...store.orders.values()].filter((o) => o.businessId === business.businessId);
  const ids = new Set(orders.map((o) => o.orderId));
  const risks = orders.flatMap((o) => store.risks.get(o.orderId) || []);
  const alerts = [...store.alerts.values()].filter((a) => ids.has(a.orderId));
  const name = (id) => (store.partners.get(id) || {}).name || id;
  return { partners, orders, risks, alerts, name };
}

function ownerText(store, business) {
  const d = gather(store, business);
  const out = ['OWNER DATA', 'Owner: ' + (business.name || business.mobile) + (business.address ? ' (' + business.address + ')' : '')];
  ['supplier', 'client', 'transporter'].forEach((role) => {
    const list = d.partners.filter((p) => p.role === role);
    out.push(role[0].toUpperCase() + role.slice(1) + 's: ' + (list.length ? list.map((p) => p.name + (role === 'supplier' ? ' (' + p.supplierType + ')' : '') + ', ' + p.address).join('; ') : 'none'));
  });
  out.push('Orders:');
  if (!d.orders.length) out.push('- none');
  d.orders.forEach((o) => {
    out.push('- ' + o.orderId + ', client ' + d.name(o.clientId) + ', due ' + o.dueDate + ', buffer ' + o.bufferDays + ' days, status ' + o.status + '. Lines: ' +
      o.lines.map((l) => d.name(l.supplierId) + ' ' + inr(l.valueRupees) + ' [' + l.status + ']').join('; '));
    (store.risks.get(o.orderId) || []).forEach((r) => out.push('  Latest risk check: ' + d.name(r.supplierId) + ', ' + r.riskLevel + ' risk, ' + inr(r.valueAtRiskRupees) + ' at risk, ' + r.delayDays + '-day delay. ' + r.reason));
  });
  out.push('Alerts: ' + (d.alerts.length ? '' : 'none'));
  d.alerts.forEach((a) => out.push('- ' + a.status + (a.chosen ? ' (chosen: ' + a.chosen + ', saved ' + inr(a.savedRupees) + ')' : '') + ': ' + a.message));
  const total = d.orders.reduce((s, o) => s + o.lines.reduce((x, l) => x + l.valueRupees, 0), 0);
  const open = d.alerts.filter((a) => a.status === 'open').reduce((s, a) => s + a.valueAtRiskRupees, 0);
  const saved = d.alerts.reduce((s, a) => s + (a.savedRupees || 0), 0);
  out.push('Totals: ' + d.orders.length + ' orders, order value ' + inr(total) + ', value at risk in open alerts ' + inr(open) + ', money saved ' + inr(saved) + '.');
  return out.join('\n');
}

function cleanHistory(history) {
  if (!Array.isArray(history)) return [];
  return history.slice(-6)
    .filter((m) => m && (m.role === 'user' || m.role === 'assistant') && typeof m.content === 'string')
    .map((m) => ({ role: m.role, content: m.content.slice(0, 800) }));
}

function register(router, store) {
  router.get('/ai/status', async () => {
    const s = await ollama.status();
    return { available: s.available, model: s.model, mode: s.available ? 'ollama' : 'basic' };
  });

  router.post('/ai/chat', async ({ body, business }) => {
    requireFields(body, ['message']);
    const message = String(body.message).trim().slice(0, 600);
    if (!message) throw new HttpError(400, 'Type a question first');

    const s = await ollama.status();
    if (s.available) {
      try {
        const messages = [{ role: 'system', content: systemPrompt(store, business, body.language) }, ...cleanHistory(body.history), { role: 'user', content: message }];
        const reply = await ollama.chat(s.model, messages);
        return { reply, source: 'ollama', model: s.model };
      } catch (e) {
        console.error('Ollama failed, using basic answer:', e.message);
        await ollama.status(true); // look again next time
      }
    }
    const d = gather(store, business);
    return { reply: basicAnswer(message, dataLines(d)), source: 'basic', model: null };
  });
}

'use strict';
// Talks to Ollama, a free AI model that runs on this laptop. Nothing is sent to the internet.
// Settings (optional): OLLAMA_URL (default http://127.0.0.1:11434) and OLLAMA_MODEL (default: pick one that is installed).
const BASE = process.env.OLLAMA_URL || 'http://127.0.0.1:11434';
const PREFERRED = ['llama3.2', 'qwen2.5', 'gemma3', 'gemma2', 'phi3', 'llama3.1', 'mistral'];

async function fetchJson(path, options, timeoutMs) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    const res = await fetch(BASE + path, { ...options, signal: controller.signal });
    if (!res.ok) throw new Error('Ollama answered ' + res.status);
    return await res.json();
  } finally {
    clearTimeout(timer);
  }
}

function pickModel(models) {
  if (process.env.OLLAMA_MODEL) return process.env.OLLAMA_MODEL;
  const chat = models.filter((m) => !/embed/i.test(m.name));
  for (const prefix of PREFERRED) {
    const hit = chat.find((m) => m.name.toLowerCase().startsWith(prefix));
    if (hit) return hit.name;
  }
  chat.sort((a, b) => (a.size || 0) - (b.size || 0));
  return chat.length ? chat[0].name : null;
}

let cache = { at: 0, info: { available: false, model: null, models: [] } };

// Is Ollama running, and which model will we use? Cached for a few seconds.
async function status(force) {
  if (!force && Date.now() - cache.at < 8000) return cache.info;
  let info;
  try {
    const tags = await fetchJson('/api/tags', {}, 1500);
    const models = tags.models || [];
    const model = pickModel(models);
    info = { available: Boolean(model), model, models: models.map((m) => m.name) };
  } catch (e) {
    info = { available: false, model: null, models: [] };
  }
  cache = { at: Date.now(), info };
  return info;
}

async function chat(model, messages) {
  const data = await fetchJson('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ model, messages, stream: false, options: { temperature: 0.2, num_predict: 350 } }),
  }, 90000);
  const text = data && data.message && data.message.content ? String(data.message.content).trim() : '';
  if (!text) throw new Error('Empty answer from Ollama');
  return text;
}

module.exports = { status, chat };

'use strict';
// M6 Risk dashboard. Owner: Utsav Pratap Singh Baghel.
// Provides: GET /dashboard
//
// This returns the numbers a dashboard needs. The target is SAP Analytics Cloud;
// until it is available, the app shows these numbers itself.
function register(router, store) {
  router.get('/dashboard', ({ business }) => {
    const orders = [...store.orders.values()].filter((o) => o.businessId === business.businessId);
    const ids = new Set(orders.map((o) => o.orderId));

    let totalOrderValueRupees = 0;
    orders.forEach((o) => o.lines.forEach((l) => { totalOrderValueRupees += l.valueRupees; }));

    let valueAtRiskRupees = 0;
    for (const [orderId, results] of store.risks) {
      if (ids.has(orderId)) results.forEach((r) => { valueAtRiskRupees += r.valueAtRiskRupees; });
    }

    const alerts = [...store.alerts.values()].filter((a) => ids.has(a.orderId));
    return {
      orders: orders.length,
      totalOrderValueRupees,
      valueAtRiskRupees,
      openAlerts: alerts.filter((a) => a.status === 'open').length,
      decisionsMade: alerts.filter((a) => a.status === 'resolved').length,
      savedRupees: alerts.reduce((sum, a) => sum + a.savedRupees, 0),
    };
  });
}
